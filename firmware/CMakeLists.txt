cmake_minimum_required(VERSION 3.22)
if(NOT DEFINED ENV{IDF_PATH})
    message(FATAL_ERROR "Activate ESP-IDF 6.1 or use tools/lab.py with the installed IDF Python.")
endif()
set(IDF_TARGET "esp32c6" CACHE STRING "Hardware target")
set(ENV{IDF_COMPONENT_MANAGER} "0")
set(PROJECT_VER "0.7.0")
get_filename_component(MESH_SYSTEM_ROOT "${CMAKE_CURRENT_LIST_DIR}/.." ABSOLUTE)
set(EXTRA_COMPONENT_DIRS "${MESH_SYSTEM_ROOT}/third_party")
# Build only the application's dependency graph.
set(COMPONENTS main)
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
if(NOT IDF_VERSION_MAJOR EQUAL 6 OR NOT IDF_VERSION_MINOR EQUAL 1)
    message(FATAL_ERROR "This project requires ESP-IDF 6.1.x.")
endif()
project(mesh_lite_node)
if(NOT IDF_TARGET STREQUAL "esp32c6")
    message(FATAL_ERROR "This project targets ESP32-C6 only.")
endif()
include(cmake/mesh_lite_idf61.cmake)

if(NOT CONFIG_MESH_LITE_MAXIMUM_LEVEL_ALLOWED EQUAL 4)
    message(FATAL_ERROR "Chain profiles require MESH_LITE_MAXIMUM_LEVEL_ALLOWED=4. Use an isolated role/device configuration.")
endif()
